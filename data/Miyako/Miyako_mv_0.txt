t.Mv_Atk_StdA <-
{
	function Init_After()
	{
		// ■レイヤー構造：通常レイヤ(1<<1) / ヒット時レイヤ(1<<2)
		BMvTbl.SetHan6LayerMask( { val = (1<<2) } ); // ヒット時レイヤーは基本的に消しておく
		
		BMvTbl.SetLP(0,0);
		BMvTbl.SetLP(1,2);
		BSound.SE_Play( { type=_SeType_Player, num=[166]} ); // ルシアンの音声
		
	}
	function Update_After()
	{
		if( BMvTbl.GetLP(0)>0 )
		{
			BMvTbl.AddLP(0,-1);
			if( BMvTbl.GetLP(0)<=0 )
			{
				BMvTbl.SetHan6LayerMask( { val = (1<<1) } ); // 通常レイヤーを消す
			}
		}
	}
	function HitInterrupt_After()
	{
		if( Battle_Std.CheckHitTiming() )
		{
			//
			BMvTbl.SetLP(0, BMvTbl.GetLP(1) );
		}
	}
	function LastUpdate_After()
	{
		BMvTbl.SetHan6LayerMask( { val = 0 } ); // レイヤーマスクを元に戻す
	}
}

local maketmpl_SkillStepPushPunch = function( param={} )
{
	local ret_tmpl = {};
	
	local mvParam =
	{
		extend = 0,
		damageShiftPos = 0,
	}

	switch( param.type )
	{
	case "A":
		mvParam.extend = { ButtonMask=(1<<0), CheckFrameID=50, EndFrameID=60, SetPattern="214A_End" };
		break;
	case "B":
		mvParam.extend = { ButtonMask=(1<<1)|(1<<2), CheckFrameID=50, EndFrameID=60, SetPattern="214B_End" };
		break;
	case "EX":
		mvParam.damageShiftPos = 60;
		break;
	case "BC":
		break;
	}
	
	ret_tmpl.flags <- def_TmplFlags_NoSurinuke;
	
	ret_tmpl.Init_After <- function()
	{
	}
	
	ret_tmpl.FrameUpdate_After <- function() : (mvParam)
	{
		local mvs = BMvTbl.GetMvStatus();
		
		if( mvParam.extend )
		{
			Battle_Std.SetPattern_NotHoldButton( mvParam.extend ); // ボタンホールドしてなかったらパターンジャンプ
		}
		
		Battle_Std.SetArmor_Param2( { Param2=4 } );//Param2&vのときにアーマー
		
		switch( Battle_Std.GetUpdateFrameID() )
		{
		case 100:
			break;
		case 200:
			// 214EXと214[A]と214[B]の攻撃タイミング
			// あたる直前に自分にかかってるヒットストップを消す（アーマーで耐えたヒットストップが残り続けることがあるバグ対策）
			// Def_Sys_CheckArmorFlag_EndParam2ArmorTiming の修正で対応済みだが一応残しておく
			local hit_stop = BMvTbl.GetHitStop();
			if( hit_stop > 0 )
			{
				BMvTbl.SetHitStop( 0, _ValSet );
			}
			break;
		}
	}
	
	ret_tmpl.HitInterrupt_After <- function() : (mvParam)
	{
		if( mvParam.damageShiftPos )
		{
			if( Battle_Std.CheckDamageTiming_ExceptDown() )
			{
				Battle_Std.SetPosition_DamageHanteiRect( { power = mvParam.damageShiftPos } );
			}
		}
	}
	
	return ret_tmpl;	
}


t.Mv_Skill_214A <- maketmpl_SkillStepPushPunch( { type="A" } );
t.Mv_Skill_214B <- maketmpl_SkillStepPushPunch( { type="B" } );
t.Mv_Skill_214EX <- maketmpl_SkillStepPushPunch( { type="EX" } );
t.Mv_Skill_214BC <- maketmpl_SkillStepPushPunch( { type="BC" } );

local maketmpl_SkillAntiAir = function( param={} )
{
	local ret_tmpl = {};
	
	local mvParam =
	{
		isMuteki = 0,
		isHeadMuteki = 0,
		zanzou = 0, // 残像で凄さを出す
		ohFlash = 0, // 中段用の発光
		enableAtkCancel = 0,
		isEX = 0,
	}

	switch( param.type )
	{
	case "A":
		mvParam.isMuteki = 1;
		ret_tmpl.fgp <- 1;
		break;
	case "B":
		mvParam.zanzou = 1;
		mvParam.ohFlash = 0;
		break;
	case "BC":
		break;
	case "EX":
		mvParam.isMuteki = 1;
		mvParam.enableAtkCancel = 1;
		ret_tmpl.fgp <- 1;
		mvParam.isEX = 1;
		break;
	}
	
	ret_tmpl.Init_After <- function() : (mvParam)
	{
		if( mvParam.isMuteki )
		{
			Battle_Std.MoveCodeEx.AddFlag( 7, def_MC7_Muteki );//無敵技
		}
		if( mvParam.enableAtkCancel )
		{
			Battle_Std.MoveCode.AddFlag( def_MC_EnableCansel_SkilltoAtk|def_MC_EnableCansel_SkilltoJump|def_MC_EnableCansel_SkilltoSkill );
		}
	}
	
	ret_tmpl.FrameUpdate_After <- function() : (mvParam)
	{
		if( mvParam.zanzou )
		{
			Battle_Std.Chr013.PowerfulAfterImageUpdate();
		}

		if( mvParam.isHeadMuteki )
		{
			Battle_Std.SetHitMuteki( 1, 8, _HitCheckFlag_Head ); // Param1 & 8 の時、_HitCheckFlag_Head 頭無敵
		}
		
		switch( Battle_Std.GetUpdateFrameID() )
		{
		case 50:
			if( mvParam.ohFlash )
			{
				// BMvEff.SetCharaFlash( { color = 0xFFF0F0, type = 0, time = 10 } );
				BMvEff.SetCharaFlash( { color = 0x4F4040, type = 0, time = 10 } );
			}
			break;
		case 100:
			break;
		}
	}
	
	ret_tmpl.HitInterrupt_After <- function() : (mvParam)
	{
		if( mvParam.isEX ) // 240122 EX版はガード時減速処理有り
		{
			if( Battle_Std.CheckGuardTiming() )
			{
				_dpn("EX版はガード時減速処理有り");
				local nowvec = BMvTbl.GetVector(0); // 現在のベクトルを得る
				local vecx = nowvec.x*70/100; // Xベクトルを減速する
				BMvTbl.SetVector( { x=vecx, flags=_Vector_Normal } );
			}
		}
	}
	
	ret_tmpl.LastUpdate_After <- function() : (mvParam)
	{
		if( mvParam.zanzou )
		{
			BMvEff.PcAfterImage_Clear();
		}
	}
	
	return ret_tmpl;	
}


t.Mv_Skill_623A <- maketmpl_SkillAntiAir( { type="A" } );
t.Mv_Skill_623B <- maketmpl_SkillAntiAir( { type="B" } );
t.Mv_Skill_623EX <- maketmpl_SkillAntiAir( { type="EX" } );
t.Mv_Skill_623BC <- maketmpl_SkillAntiAir( { type="BC" } );

		mvParam.useAirCountSlot = 1;
		break;
	}
	
	ret_tmpl.Init_After <- function() : (mvParam)
	{
		if( mvParam.useAirCountSlot == 1 )

			BMvTbl.AddAirJumpCount( -10 );
			BMvTbl.SetAirJumpOK(1);  //再度ジャンプできないようにする
		}
	}
	return ret_tmpl;	
}

local maketmpl_SkillAirFallingElbow = function( param={} )
{
	local ret_tmpl = {};
	
	local mvParam =
	{
		ball = 0,
	}

	switch( param.type )
	{
	case "B":
		mvParam.ball = { mv="Mv_FireBall_J4BC", pat="RcJ4BC" };
		break;
	}
	
	ret_tmpl.FrameUpdate_After <- function() : (mvParam)
	{
		switch( Battle_Std.GetUpdateFrameID() )
		{
		case 500:
			if( mvParam.ball )
			{
				local ball = Battle_Std.CreateFireBall( mvParam.ball );
				if( ball.push() )
				{
					BMvEff.SetObjectFlags( { flags=_ObjFlags_ToParentHitBack } );
					ball.pop();
				}
			}
			break;
		}
	}
	
	return ret_tmpl;	
}
