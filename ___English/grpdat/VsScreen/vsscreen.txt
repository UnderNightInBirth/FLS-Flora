function Scene_VsScreenFunc::Update()
{
	local p = Scene_VsScreenFunc;

	local fr;

	if( p.IsDone != 0 )
	{
		// ìríÜÇ≈è¡Ç≥Ç»Ç¢Ç∆Ç¢ÇØÇ»Ç¢Ç‡ÇÃ

		{
			fr = p.PA_Manager.GetCAnime( "S1_word" );
			fr.SetView(0);
			fr = p.PA_Manager.GetCAnime( "S1_bg" );
			fr.SetView(0);
			fr = p.PA_Manager.GetCAnime( "S1_obj" );
			fr.SetView(0);
		}
		else
		if( p.Count == 181 )
		{
			fr = p.PA_Manager.GetCAnime( "S2_bg" );
			fr.SetView(0);
			fr = p.PA_Manager.GetCAnime( "S2_1P" );
			fr.SetView(0);
		}
		else
		if( p.Count == 241 )
		{
			fr = p.PA_Manager.GetCAnime( "S3_bg" );
			fr.SetView(0);
			fr = p.PA_Manager.GetCAnime( "S3_2P" );
			fr.SetView(0);
		}
		else
		if( p.Count == 301 )
		{
			fr = p.PA_Manager.GetCAnime( "S4_bg_a" );
			fr.SetView(0);
		}
	}

	p.PA_Manager.Update();
	p.Count++;
}
