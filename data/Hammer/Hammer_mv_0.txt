	{

			{

				enemy.pop();
			}
		}
		
		BMvTbl.SetLP(30,0)

	}
	function Update()
	{
	
		local s = BMvTbl.GetMvStatus();
		if (s.FrameID != 5 && s.FrameID != 10 && s.FrameID != 200 && s.FrameID != 250)
		{

			local captured = ( BMvTbl.GetMvStatus().FrameID==50 && BMvTbl.GetLP(30) );
			if(captured)
			{
				local enemy = BMvCore.GetNearEnemyCharaData();
				if( enemy.push() )
				{
					Battle_Std.InitVector();
					enemy.pop();
				}
			}
			
			local enemy = BMvCore.GetNearEnemyCharaData();
			if(enemy.push())
			{
				enemy.pop()
			}
			
			local stick = BMvTbl.GetStickHold();
			
			local vec_x = 0;
			local vec_y = 0;
			
			if( stick == 9 || stick == 6 || stick == 3 ) vec_x = -1000;
			if( stick == 7 || stick == 4 || stick == 1 ) vec_x = 1000;
			if( stick == 7 || stick == 8 || stick == 9 ) vec_y = -1000;
			if( stick == 1 || stick == 2 || stick == 3 ) vec_y = 1000;
			
			if( !captured ) vec_y = 0;
			
			
			local enemy = BMvCore.GetNearEnemyCharaData();
			if( enemy.push() )
			{
				local airtime = ( BMvTbl.GetMvName() == "Mv_Jump_F" || BMvTbl.GetMvName() == "Mv_Jump_B" || BMvTbl.GetMvName() == "Mv_Jump_N" );
	
				local bs = BtlMvStd.GetBoundStatus();
				// BMvTbl.SetPattern(313);
	
				// BMvTbl.SetVector( { x=vec_x, addx=0, y=vec_y, addy=0, flags=_Vector_Add } );
				if(captured)
				{
					BMvTbl.SetVector( { x=vec_x, y=vec_y, flags=_Vector_Normal } );
				}
				
				if(!captured)
				{
					if(BCMDTbl.CheckPosState( _PosState_Air ))
					{
						BMvTbl.SetVector( { x=vec_x/16, flags=_Vector_Normal|_VecFlag_Add } );
					}
					if(BCMDTbl.CheckPosState( _PosState_Ground ))
					{
						BMvTbl.SetVector( { x=vec_x, flags=_Vector_Normal } );
					}
				}
				
				// BMvTbl.SetVector( { x=vec_x, addx=0, y=vec_y, addy=0, flags=_Vector_Normal } );
				
				
				enemy.pop();
			}
		}
		
	}

					BMvTbl.JumpFrameID(22);

				}
				break;

	}

				BMvEff.Liberate_Set();
				BMvEff.Liberate_SetType( _LiberateType_Normal ); 
			}
			
			local EXmvs = Battle_Std.IsMatchMvNameArray( ["Mv_Skill_236EX","Mv_Skill_623EX"] ) // There is probably a better way to do this
			if( gaugeType==1 && EXmvs )
			{
				BMvTbl.SetLP(91,1); // We DidEx
			}
			
			if( exs==0 && BMvTbl.GetLP(90)==1 && BMvTbl.GetLP(91)==0 ) // Check we didnt EX and were out of HEAT
			{
				BMvEff.SetSpGauge( { value=5000 } );
				BMvTbl.SetLP(90,0);
				BMvTbl.SetLP(91,0);
			}
		}
    }
}


local set_JumpCtrlVector = function()
{
	local vec = BMvTbl.GetVector();
	
	if( (vec.y + vec.addy ) >= 0 )
	{
		//print("\n .");//23F
		return; // ‰º~’†‚Í¸”s
	}
	
	local x = 0; //vector amount
	local plus_x = 96; //move by this every frame
	//command‚¾‚Æ‘Šè•ûŒüŠî€‚©H
	if( ( BMvTbl.CheckStickHold( (1<<4) | (1<<7) | (1<<1) ) ) && vec.x > -800 ) x -= plus_x;
	if( ( BMvTbl.CheckStickHold( (1<<6) | (1<<9) | (1<<3) ) ) && vec.x <  800 ) x += plus_x;
	
	if( x )
	{
		vec.x += x;
		BMvTbl.SetVector( { x=vec.x, flags=_Vector_Normal } );
	}
}
