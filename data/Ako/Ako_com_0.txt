// Section: Smart Steer

{   // Skill 45: "A" button mash combo
    stmp = com.GetSkill(45); // Create skill slot 45
    stmp.SetLotWait( 320 ); // Wait time after finishing/failing this skill
    stmp.SetLotFailedAvg( 10 ); // Probability of failing this skill

    // Condition function for this skill
    stmp.checkfunc = function()
    {
        local st = BCom.GetStatus(); // Get current status
        if( !st.IsPMoveable() ) return 0; // Can't act? Don't use this skill
        if( st.GetELenX() > 40000 ) return 0; // Too far from opponent? Don't use
        if( !st.CheckPosState( _PosState_Ground ) ) return 0; // Not on the ground? Don't use
        if( st.GetELenY() != 0 ) return 0; // Opponent must also be on the ground

        return com_frequency2; // Otherwise, use global frequency
    }
    stmp.updatefunc = function() // Called when this skill is chosen
    {
        return 1; // Always return 1 (success)
    }
    // Add smart steer
    stmp.PlusTagChainCombo( "SSCombo", [ "A",{rapid=1},"A","A","A","A",{rapid=1},["A",5] ] );
}
