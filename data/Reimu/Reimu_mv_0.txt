t.Mv_Skill_IWEXIST <-
{
	flags = def_TmplFlags_NoAddComboRate // Don't add to combo
    function HitInterrupt_After()
    {
        Battle_Std.SetThrowHitFinalize(256);
		BMvTbl.SetLP(11, 0);
    }
    function Finalize()
    {
        Battle_Std.SwitchNextMoveTable("Mv_Neutral",[256,"Mv_Skill_IWEXIST_Hit"]);
    }
}

t.Mv_Skill_IWEXIST_Hit <-
{
	flags = def_TmplFlags_NoAddComboRate // Don't add to combo
	function Init_After()
	{
		BMvTbl.SetLP(10, 1); // Set Counter
		BMvTbl.SetLP(11, 0);
		BMvEff.FadeProc_SetRenderFlag(0); // 0:カットインBG描画なし

		BMvEff.SetStopTime( { time=60, stopme=0 } );
	}
	function FrameUpdate_After()
	{
		switch( Battle_Std.GetUpdateFrameID() )
		{
		case 10:
			local cnt = BMvTbl.GetLP(10);
			Battle_Std.CreateObjectEX( { x=256, y=-128, mvname="Mv_Obj_YinYang", pat="YinYang", LP={ slot=0, val=cnt }, flags=_Position_ToolShift } );
			BMvTbl.AddLP(10, 1); // Increment Counter
			break;
		}
	}
	function Finalize_After()
	{

}
