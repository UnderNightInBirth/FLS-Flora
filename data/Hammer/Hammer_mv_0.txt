t.Mv_Skill_63214EX <-
{
	function Init_After()
	{
		BMvTbl.SetLP(30,0); // reset capture
	}
	function FrameUpdate_After()
	{
		// Battle_Std.JumpFrameID_NotHoldButton({ mask=(1<<2), checkid=50, jumpid=60 });
        Battle_Std.SetPattern_NotHoldButton( { ButtonMask=(1<<2), CheckFrameID=50, EndFrameID=60, SetPattern="623B_End" } );

	    local s = BMvTbl.GetMvStatus();
        if(s.MvCount==30)
        {
							// local alpha = BMvTbl.GetDrawAlpha();
			local boing = Battle_Std.CreateObjectEX( { x=0, y=0, pat="BigLimit", objectflags=_ObjFlags_NoGround|_ObjFlags_ToParentStop } );
			if(boing.push())
			{
				BMvEff.SetObjectFlags({ flags=_ObjFlags_EraseParentPatChange|_ObjFlags_ToParentHitStatus|_ObjFlags_EraseParentDamage });
				BMvTbl.SetScale({ x=60000, y=60000 });
				Battle_Std.SetPos_MarkingPlayer();
				boing.pop();
			}
		}
		if(s.FrameID==250 && BMvTbl.GetLP(30))
        {
			local enemy = BMvCore.GetNearEnemyCharaData();
			if( enemy.push() )
			{
				BMvTbl.SetVector( { x=0, addx=0, y=0, addy=128, flags=_Vector_Normal|_VecFlag_Add } );
				enemy.pop();
			}
			BMvTbl.SetLP(30,0)
		}
		
		if(s.MvCount==160)
		{
			BMvEff.AttackInfoString_Set({ word = "UNBLOCKABLE" });
			// BMvEff.AttackInfoString_Set({ word = s.MvCount });
		}
	
	}
	function Finalize_After()
	{
	
		if(BMvTbl.GetLP(30))
        {
			local enemy = BMvCore.GetNearEnemyCharaData();
			if( enemy.push() )
			{
				BMvTbl.SetVector( { x=0, addx=0, y=0, addy=128, flags=_Vector_Normal|_VecFlag_Add } );
				enemy.pop();
			}
		}
		
		BMvTbl.SetLP(30,0)

	}
	function Update()
	{
	
		local s = BMvTbl.GetMvStatus();
		if (s.FrameID != 5 && s.FrameID != 10 && s.FrameID != 200 && s.FrameID != 250)
		{

			local captured = ( BMvTbl.GetMvStatus().FrameID==50 && BMvTbl.GetLP(30) );
			if(captured)
			{
				local enemy = BMvCore.GetNearEnemyCharaData();
				if( enemy.push() )
				{
					Battle_Std.InitVector();
					enemy.pop();
				}
			}
			
			local enemy = BMvCore.GetNearEnemyCharaData();
			if(enemy.push())
			{
				enemy.pop()
			}
			
			local stick = BMvTbl.GetStickHold();
			
			local vec_x = 0;
			local vec_y = 0;
			
			if( stick == 9 || stick == 6 || stick == 3 ) vec_x = -1000;
			if( stick == 7 || stick == 4 || stick == 1 ) vec_x = 1000;
			if( stick == 7 || stick == 8 || stick == 9 ) vec_y = -1000;
			if( stick == 1 || stick == 2 || stick == 3 ) vec_y = 1000;
			
			if( !captured ) vec_y = 0;
			
			
			local enemy = BMvCore.GetNearEnemyCharaData();
			if( enemy.push() )
			{
				local airtime = ( BMvTbl.GetMvName() == "Mv_Jump_F" || BMvTbl.GetMvName() == "Mv_Jump_B" || BMvTbl.GetMvName() == "Mv_Jump_N" );
	
				local bs = BtlMvStd.GetBoundStatus();
				// BMvTbl.SetPattern(313);
	
				// BMvTbl.SetVector( { x=vec_x, addx=0, y=vec_y, addy=0, flags=_Vector_Add } );
				if(captured)
				{
					BMvTbl.SetVector( { x=vec_x, y=vec_y, flags=_Vector_Normal } );
				}
				
				if(!captured)
				{
					if(BCMDTbl.CheckPosState( _PosState_Air ))
					{
						BMvTbl.SetVector( { x=vec_x/16, flags=_Vector_Normal|_VecFlag_Add } );
					}
					if(BCMDTbl.CheckPosState( _PosState_Ground ))
					{
						BMvTbl.SetVector( { x=vec_x, flags=_Vector_Normal } );
					}
				}
				
				// BMvTbl.SetVector( { x=vec_x, addx=0, y=vec_y, addy=0, flags=_Vector_Normal } );
				
				
				enemy.pop();
			}
		}
		
	}

	{
		if( Battle_Std.CheckHitTiming_FrameID(10) )

			if( enemy.push() )
			{
				Battle_Std.InitVector();
				enemy.pop();
			}

	}

t.Mv_Obj_Catto <-
{	
	function Init_Before()
	{
		BMvTbl.SetLP(0,0); // Initialize Player Hit Flag
	}
	function FrameUpdate_After()
	{
		local s = BMvTbl.GetMvStatus();
		local muki = BMvTbl.GetMuki();
		
		if(Battle_Std.CheckPlayerisDamage())
		{
			BMvTbl.SetLP(0,1); // Turns on Flag if player is hit
		}

		switch( s.FrameID )
		{
			case 0:	
				if ( ( Battle_Std.GetEnemyDistance()/128 <= 650 ) && ( Battle_Std.GetHaimenGamenHajiDistance() / 128 >= 32 ) && !BMvTbl.GetLP(0) )
				{
					BMvTbl.JumpFrameID(22);
				}
				break;
			
			case 22:
				BMvTbl.SetVector({ x=1024, y=-512, addx=0, addy=80, flags=_Vector_Normal  });
					// BMvEff.CreateObject( { x=0, y=-300, start_pat=502, mvname="Mv_Obj_Pot", flags=_Position_ToolShift } );
				break;
			case 30:
				if (BMvTbl.GetPosition().y >= 4000)
				{
					BMvTbl.JumpFrameID(40);
					Battle_Std.InitVector();
				}
				break;
		}
		
		if( BMvTbl.GetLP(0) )
		{
			local alpha = BMvTbl.GetDrawAlpha() - 1;				
			BMvTbl.SetDrawAlpha( { val=alpha, time=5 } );
		}
	}
}

local set_JumpCtrlVector = function()
{
	local vec = BMvTbl.GetVector();
	
	if( (vec.y + vec.addy ) >= 0 )
	{
		//print("\n .");//23F
		return; // 下降中は失敗
	}
	
	local x = 0; //vector amount
	local plus_x = 96; //move by this every frame
	//commandだと相手方向基準か？
	if( ( BMvTbl.CheckStickHold( (1<<4) | (1<<7) | (1<<1) ) ) && vec.x > -800 ) x -= plus_x;
	if( ( BMvTbl.CheckStickHold( (1<<6) | (1<<9) | (1<<3) ) ) && vec.x <  800 ) x += plus_x;
	
	if( x )
	{
		vec.x += x;
		BMvTbl.SetVector( { x=vec.x, flags=_Vector_Normal } );
	}
}
